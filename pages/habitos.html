// Lógica da página de hábitos: nome, criar, editar, excluir e concluir. const CATEGORIAS = ['Estudos', 'Autocuidado', 'Organização', 'Bem-estar']; // --- Dados --- function dataDeHoje() { return new Date().toLocaleDateString('sv-SE'); // formato AAAA-MM-DD } function listarHabitos() { return armazenamento.ler('habitos', []); } function listarConcluidosDeHoje() { const registros = armazenamento.ler('registros', {}); return registros[dataDeHoje()] || []; } function salvarConcluidosDeHoje(ids) { const registros = armazenamento.ler('registros', {}); registros[dataDeHoje()] = ids; armazenamento.salvar('registros', registros); } // --- Ações --- function criarHabito(nome, categoria) { const habitos = listarHabitos(); habitos.push({ id: Date.now(), nome: nome, categoria: categoria }); armazenamento.salvar('habitos', habitos); } function editarHabito(id, novoNome) { const habitos = listarHabitos().map(function (habito) { return habito.id === id ? { ...habito, nome: novoNome } : habito; }); armazenamento.salvar('habitos', habitos); } function excluirHabito(id) { armazenamento.salvar('habitos', listarHabitos().filter(function (h) { return h.id !== id; })); salvarConcluidosDeHoje(listarConcluidosDeHoje().filter(function (i) { return i !== id; })); } function alternarConclusao(id) { const concluidos = listarConcluidosDeHoje(); const novos = concluidos.includes(id) ? concluidos.filter(function (i) { return i !== id; }) : concluidos.concat(id); salvarConcluidosDeHoje(novos); } // --- Tela --- const elemento = function (id) { return document.getElementById(id); }; function mostrarMensagem(texto, tipo) { const caixa = elemento('mensagem'); caixa.textContent = texto; caixa.className = 'mensagem mensagem-' + tipo; setTimeout(function () { caixa.className = 'mensagem'; caixa.textContent = ''; }, 3000); } function desenharSaudacao() { const nome = armazenamento.ler('nome', ''); elemento('form-nome').hidden = nome !== ''; elemento('painel').hidden = nome === ''; elemento('saudacao').textContent = nome ? 'Olá, ' + nome + '!' : 'Bem-vinda ao Bloom'; elemento('data-hoje').textContent = new Date().toLocaleDateString('pt-BR', { weekday: 'long', day: 'numeric', month: 'long' }); } function desenharLista() { const habitos = listarHabitos(); const concluidos = listarConcluidosDeHoje(); const lista = elemento('lista-habitos'); lista.innerHTML = ''; habitos.forEach(function (habito) { const feito = concluidos.includes(habito.id); const item = document.createElement('li'); item.className = 'habito' + (feito ? ' habito-feito' : ''); const caixa = document.createElement('input'); caixa.type = 'checkbox'; caixa.checked = feito; caixa.setAttribute('aria-label', 'Concluir ' + habito.nome); caixa.addEventListener('change', function () { alternarConclusao(habito.id); desenharTudo(); }); const texto = document.createElement('span'); texto.className = 'habito-nome'; texto.textContent = habito.nome; const etiqueta = document.createElement('small'); etiqueta.textContent = habito.categoria; const botaoEditar = document.createElement('button'); botaoEditar.textContent = 'Editar'; botaoEditar.className = 'botao-pequeno'; botaoEditar.addEventListener('click', function () { const novoNome = prompt('Novo nome do hábito:', habito.nome); if (novoNome && novoNome.trim()) { editarHabito(habito.id, novoNome.trim()); mostrarMensagem('Hábito atualizado.', 'sucesso'); desenharTudo(); } }); const botaoExcluir = document.createElement('button'); botaoExcluir.textContent = 'Excluir'; botaoExcluir.className = 'botao-pequeno botao-perigo'; botaoExcluir.addEventListener('click', function () { if (confirm('Excluir o hábito "' + habito.nome + '"?')) { excluirHabito(habito.id); mostrarMensagem('Hábito excluído.', 'sucesso'); desenharTudo(); } }); item.append(caixa, texto, etiqueta, botaoEditar, botaoExcluir); lista.appendChild(item); }); elemento('lista-vazia').hidden = habitos.length > 0; } function desenharProgresso() { const total = listarHabitos().length; const feitos = listarConcluidosDeHoje().length; const porcentagem = total === 0 ? 0 : Math.round((feitos / total) * 100); elemento('progresso-texto').textContent = feitos + ' de ' + total + ' hábitos concluídos hoje'; elemento('progresso-barra').style.width = porcentagem + '%'; } function desenharTudo() { desenharSaudacao(); desenharLista(); desenharProgresso(); } // --- Eventos --- elemento('form-nome').addEventListener('submit', function (evento) { evento.preventDefault(); const nome = elemento('campo-nome').value.trim(); if (nome.length < 2) { mostrarMensagem('Digite um nome com pelo menos 2 letras.', 'erro'); return; } armazenamento.salvar('nome', nome); desenharTudo(); }); elemento('form-habito').addEventListener('submit', function (evento) { evento.preventDefault(); const nome = elemento('campo-habito').value.trim(); if (nome === '') { mostrarMensagem('Digite o nome do hábito.', 'erro'); return; } criarHabito(nome, elemento('campo-categoria').value); elemento('campo-habito').value = ''; mostrarMensagem('Hábito criado!', 'sucesso'); desenharTudo(); }); CATEGORIAS.forEach(function (categoria) { const opcao = document.createElement('option'); opcao.textContent = categoria; elemento('campo-categoria').appendChild(opcao); }); desenharTudo();
